Multisim 神了。打开之后提示无法打开 Master Database。

### 数据库文件

Multisim 的数据库文件在：

```text
%ProgramData%\National Instruments\Circuit Design Suite\
```

检查发现这个玩意是没有损坏的，但是它打不开。删除 `config` 然后重新注册 `dao360.dll` 实测有概率打得开。

### 解决方法

```bat
@echo off
setlocal enabledelayedexpansion

:: 1. Request Administrator Privileges
>nul 2>&1 "%SYSTEMROOT%\system32\cacls.exe" "%SYSTEMROOT%\system32\config\system"
if '%errorlevel%' NEQ '0' (
    echo Set UAC = CreateObject^("Shell.Application"^) > "%temp%\getadmin.vbs"
    echo UAC.ShellExecute "%~s0", "", "", "runas", 1 >> "%temp%\getadmin.vbs"
    "%temp%\getadmin.vbs"
    exit /B
)
if exist "%temp%\getadmin.vbs" del "%temp%\getadmin.vbs"
cd /d "%~dp0"

echo [1/4] Killing running processes and stopping services...
taskkill /F /IM multisim.exe >nul 2>&1
taskkill /F /IM ultiboard.exe >nul 2>&1
net stop "National Instruments Device Service" >nul 2>&1
timeout /t 1 /nobreak >nul

echo [2/4] Clearing user configuration cache...
set "VER=14.3"
set "USER_CONF=%AppData%\National Instruments\Circuit Design Suite\%VER%\config"
set "LOCAL_DATA=%LocalAppData%\National Instruments\Circuit Design Suite\%VER%"

if exist "%USER_CONF%" rd /s /q "%USER_CONF%"
if exist "%LOCAL_DATA%" rd /s /q "%LOCAL_DATA%"

echo [3/4] Resetting database file permissions and attributes...
set "DB_PATH=%ProgramData%\National Instruments\Circuit Design Suite\%VER%\database"
if exist "%DB_PATH%" (
    attrib -r -s -h "%DB_PATH%\*.*" /s /d >nul 2>&1
)

echo [4/4] Re-registering COM database engines (DAO / Jet)...
:: Register 32-bit DAO engine
if exist "%CommonProgramFiles(x86)%\Microsoft Shared\DAO\dao360.dll" (
    regsvr32 /s "%CommonProgramFiles(x86)%\Microsoft Shared\DAO\dao360.dll"
) else if exist "%CommonProgramFiles%\Microsoft Shared\DAO\dao360.dll" (
    regsvr32 /s "%CommonProgramFiles%\Microsoft Shared\DAO\dao360.dll"
)

:: Register Jet DLLs
if exist "%SystemRoot%\SysWOW64\msjet40.dll" (
    regsvr32 /s "%SystemRoot%\SysWOW64\msjet40.dll"
    regsvr32 /s "%SystemRoot%\SysWOW64\msjtes40.dll"
) else (
    regsvr32 /s "%SystemRoot%\System32\msjet40.dll"
    regsvr32 /s "%SystemRoot%\System32\msjtes40.dll"
)

echo.
echo ========================================
echo Fix completed! Please try launching Multisim.
echo If it still fails, check whether CPCompDev.prz in the database folder is 0 KB.
echo ========================================
pause
```
